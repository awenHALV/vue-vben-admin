## 页面内部 wujie 页面跳转

```wujie/jump.js
export const wujieVueJump = (routeInfo: jumpPath | jumpByWujie) => {
  // eslint-disable-next-line no-underscore-dangle
  if (!window.__POWERED_BY_WUJIE__) {
    console.error('本次路由跳转需要在无界中进行');
    return false;
  }
  if (Object.prototype.hasOwnProperty.call(routeInfo, 'targetProjectCode')) {
    const {
      targetAppCode,
      targetProjectCode,
      dynamicRoute,
      targetPath,
      dynamicRoutePath = '',
      queryMenu = {},
      query = {},
    } = routeInfo;

    window.$wujie?.bus.$emit(JUMPROUTE_EVENT, {
      projectCode: website.projectCode,
      targetAppCode,
      targetProjectCode,
      dynamicRoute,
      targetPath,
      dynamicRoutePath,
      queryMenu,
      query,
    });
  } else {
    // 新的方式
    const {
      targetAppCode,
      path,
      forceWindow = false,
      isJumpBase = false,
      query = {},
      option = {},
    } = routeInfo;

    if (!targetAppCode) {
      console.error('targetAppCode必传');
      return false;
    }
    if (!path) {
      console.error('path必传');
      return false;
    }
    window.$wujie?.bus.$emit(JUMPROUTE_EVENT, {
      targetAppCode,
      path,
      forceWindow,
      isJumpBase,
      query,
      option,
    });
  }
  return true;
};
```

## main.ts

### 引入 wujie 的事件

```js
import {
  ROUTERCHANGE_EVENT,
  CHANGETHEME_EVENT,
  NOTICECHILDAPPTOKEN_EVENT,
  JUMPROUTESTATIC_EVENT,
  DEACTIVATEDAPP,
  CHANGELANUAGE_EVENT,
} from '@/wujie/event';

async function init() {
  if (isLogin()) {
    /* 页面刷新后，加载动态路由 */
    const appStore = useAppStore();
    if (appStore.menuFromServer) {
      await appStore.fetchServerMenuConfig();
      setup();
    }
  } else {
    setup();
  }
}

// 嵌入子应用后,获取菜单数据
if (window.__POWERED_BY_WUJIE__) {
  const { menuData, btns } = window.$wujie?.props;
  window.__WUJIE_MOUNT = async () => {
    // 更新token
    updateToken();
    const appStore = useAppStore();
    // 获取子应用-external/private/mine/app
    appStore.getApps();
    // 将获取到的菜单添加到当前路由
    appStore.addRouteMenuData(menuData, btns);
    setup();
    // 切换路由事件
    window.$wujie?.bus.$on(ROUTERCHANGE_EVENT, routerChangeListener);
    // 主题切换
    window.$wujie?.bus.$on(CHANGETHEME_EVENT, changeTheme);
    // 中英文切换
    window.$wujie?.bus.$on(CHANGELANUAGE_EVENT, changeLanguage);
    // 子应用更新token
    window.$wujie?.bus.$on(NOTICECHILDAPPTOKEN_EVENT, updateToken);
    // 子应用跳转到本地静态路由
    window.$wujie?.bus.$on(JUMPROUTESTATIC_EVENT, jumpStaticRoute);
    window.$wujie?.bus.$on(DEACTIVATEDAPP, deactivatedApp);
  };
  window.__WUJIE.mount();
  window.addEventListener('resize', () => {
    window.innerWidth = window.top.innerWidth;
  });
} else {
  // 子项目通过接口获取菜单
  init();
}
```
